cmake_minimum_required(VERSION 3.24)
project(AlloyCocoaAccessibilityPatch LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)

find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core Gui CorePrivate GuiPrivate)
if(NOT QT_FEATURE_accessibility)
    message(FATAL_ERROR "The installed Qt build has accessibility disabled")
endif()
if(NOT EXISTS "${QTBASE_SOURCE_DIR}/src/plugins/platforms/cocoa/main.mm")
    message(FATAL_ERROR "QTBASE_SOURCE_DIR must be the extracted Qt 6.11.2 source tree")
endif()

set(cocoa "${QTBASE_SOURCE_DIR}/src/plugins/platforms/cocoa")
file(GLOB sources "${cocoa}/*.mm" "${cocoa}/*.cpp" "${cocoa}/*.h")
# These files are textually included by qnsview.mm in the upstream target.
list(FILTER sources EXCLUDE REGEX "/qnsview_[^/]+\\.mm$")
if(NOT QT_FEATURE_opengl)
    list(FILTER sources EXCLUDE REGEX "qcocoaglcontext")
endif()
if(NOT QT_FEATURE_vulkan)
    list(FILTER sources EXCLUDE REGEX "qcocoavulkaninstance")
endif()
if(NOT QT_FEATURE_sessionmanager)
    list(FILTER sources EXCLUDE REGEX "qcocoasessionmanager")
endif()

add_library(qcocoa MODULE ${sources})
target_compile_definitions(qcocoa PRIVATE QT_PLUGIN)
target_include_directories(qcocoa PRIVATE "${cocoa}")
target_link_libraries(qcocoa PRIVATE Qt6::CorePrivate Qt6::GuiPrivate)
foreach(framework Foundation AppKit Carbon CoreServices CoreVideo IOKit IOSurface Metal QuartzCore UniformTypeIdentifiers)
    find_library(fw_${framework} ${framework} REQUIRED)
    target_link_libraries(qcocoa PRIVATE ${fw_${framework}})
endforeach()
qt_add_resources(qcocoa qcocoaresources PREFIX "/qt-project.org/mac/cursors" BASE "${cocoa}"
    FILES "${cocoa}/images/sizeallcursor.png" "${cocoa}/images/spincursor.png" "${cocoa}/images/waitcursor.png")
set_target_properties(qcocoa PROPERTIES SUFFIX ".dylib" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/platforms")
