if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.24)
    project(AlloyLocalControl VERSION 0.1.0 LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_AUTOMOC ON)
    find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core Network)
    include(CTest)
endif()
add_library(alloy_local_security STATIC PrivateDirectory.cpp PrivateDirectory.h)
target_link_libraries(alloy_local_security PUBLIC Qt6::Core)
target_include_directories(alloy_local_security PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(WIN32)
    target_link_libraries(alloy_local_security PRIVATE advapi32)
endif()
add_executable(alloy_local_client client.cpp)
set_target_properties(alloy_local_client PROPERTIES OUTPUT_NAME alloy-local-client)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set_target_properties(alloy_local_client PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endif()
target_link_libraries(alloy_local_client PRIVATE Qt6::Core Qt6::Network)
if(BUILD_TESTING)
    find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Test)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(local_transport_test transport_test.cpp)
    target_link_libraries(local_transport_test PRIVATE Qt6::Core Qt6::Network Qt6::Test alloy_local_security)
    add_test(NAME local_transport COMMAND local_transport_test $<TARGET_FILE:alloy_local_client> "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../alloyctl.py")
endif()
