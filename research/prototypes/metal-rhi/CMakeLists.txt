cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(AlloyMetalSpike LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    set(CMAKE_AUTOMOC ON)
endif()
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core GuiPrivate Widgets)
find_package(OpenCASCADE 7.9.3 EXACT REQUIRED COMPONENTS FoundationClasses ModelingData ModelingAlgorithms)
qt_add_executable(metal_rhi_spike MACOSX_BUNDLE main.cpp)
target_include_directories(metal_rhi_spike SYSTEM PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_link_libraries(metal_rhi_spike PRIVATE Qt6::Core Qt6::Gui Qt6::GuiPrivate Qt6::Widgets
    TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim TKMesh)
target_compile_definitions(metal_rhi_spike PRIVATE ALLOY_OCCT_VERSION="7.9.3")
set_target_properties(metal_rhi_spike PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.alloycad.MetalSpike MACOSX_BUNDLE_BUNDLE_NAME "Alloy Metal Spike")
set_source_files_properties(shaders/color.vert.qsb PROPERTIES QT_RESOURCE_ALIAS color.vert.qsb)
set_source_files_properties(shaders/color.frag.qsb PROPERTIES QT_RESOURCE_ALIAS color.frag.qsb)
qt_add_resources(metal_rhi_spike "pinned_shader_bundles" PREFIX "/metal-rhi" FILES shaders/color.vert.qsb shaders/color.frag.qsb)
